#!/bin/bash

adduser profesor1
adduser alumno1
adduser alumno2
adduser alumno3
adduser alumno4

groupadd grupo_mañana
groupadd grupo_tarde
groupadd grupo_profesor

usermod -aG grupo_mañana alumno1
usermod -aG grupo_mañana alumno2
usermod -aG grupo_tarde alumno3
usermod -aG grupo_tarde alumno4

usermod -aG grupo_mañana profesor1
usermod -aG grupo_tarde profesor1
usermod -aG grupo_profesor profesor1

sudo mkdir /home/alumno1/PracticasIndividuales
#ponemos chown porque al crearlo con sudo el propietario seria root. De esta manera especificamos que el alumno1 es el propietario.
sudo chown alumno1:alumno1 /home/alumno1/PracticasIndividuales

sudo mkdir /home/alumno2/PracticasIndividuales
sudo chown alumno2:alumno2 /home/alumno2/PracticasIndividuales

sudo mkdir /home/alumno3/PracticasIndividuales
sudo chown alumno3:alumno3 /home/alumno3/PracticasIndividuales

sudo mkdir /home/alumno4/PracticasIndividuales
#Añade en /home un directorio para cada grupo creado, que será el que se utilice para los trabajos en equipo. PracticasIndividuales
sudo chown alumno4:alumno4 /home/alumno4/PracticasIndividuales

#hacemos esto porque sino el profesor perteneceria a la categoría others y others tiene permisos nulo (000). Entonces se crea un grupo extra donde se introduce al profesor para que de esta manera tenga permisos de grupo.
sudo chown alumno1:grupo_profesor /home/alumno1/PracticasIndividuales

sudo chown alumno2:grupo_profesor /home/alumno2/PracticasIndividuales

sudo chown alumno3:grupo_profesor /home/alumno3/PracticasIndividuales

sudo chown alumno4:grupo_profesor /home/alumno4/PracticasIndividuales


sudo chmod 710 /home/alumno1/PracticasIndividuales
sudo chmod 710 /home/alumno2/PracticasIndividuales
sudo chmod 710 /home/alumno3/PracticasIndividuales
sudo chmod 710 /home/alumno4/PracticasIndividuales

sudo touch /home/alumno1/PracticasIndividuales/ejercicio1
sudo touch /home/alumno2/PracticasIndividuales/ejercicio1
sudo touch /home/alumno3/PracticasIndividuales/ejercicio1
sudo touch /home/alumno4/PracticasIndividuales/ejercicio1

sudo touch /home/alumno1/PracticasIndividuales/notasEjercicio1
sudo touch /home/alumno2/PracticasIndividuales/notasEjercicio1
sudo touch /home/alumno3/PracticasIndividuales/notasEjercicio1
sudo touch /home/alumno4/PracticasIndividuales/notasEjercicio1

sudo chmod  660 /home/alumno1/PracticasIndividuales/ejercicio1
sudo chmod  660 /home/alumno2/PracticasIndividuales/ejercicio1
sudo chmod  660 /home/alumno3/PracticasIndividuales/ejercicio1
sudo chmod  660 /home/alumno4/PracticasIndividuales/ejercicio1


sudo chmod  640 /home/alumno1/PracticasIndividuales/notasEjercicio1
sudo chmod  640 /home/alumno2/PracticasIndividuales/notasEjercicio1
sudo chmod  640 /home/alumno3/PracticasIndividuales/notasEjercicio1
sudo chmod  640 /home/alumno4/PracticasIndividuales/notasEjercicio1


sudo mkdir /home/grupo_mañana
sudo mkdir /home/grupo_tarde


sudo touch /home/grupo_mañana/archivo.txt
sudo touch /home/grupo_mañana/archivo.sh
sudo touch /home/grupo_tarde/archivo.txt
sudo touch /home/grupo_tarde/archivo.sh


#-R lo aplica de manera recursiva a todo. Si la quitamos solo lo aplicaria al propietario/grupo. Al ponerlo tambien se aplica sobre archivo.txt y archivo.sh
sudo chown -R profesor:grupo_mañana /home/grupo_mañana
sudo chown -R profesor:grupo_tarde /home/grupo_tarde

sudo chmod 771 /home/grupo_mañana
sudo chmod 771 /home/grupo_tarde
sudo chmod 771 /home/grupo_mañana/archivo.sh
sudo chmod 771 /home/grupo_tarde/archivo.sh
sudo chmod 761 /home/grupo_mañana/archivo.txt
sudo chmod 761 /home/grupo_tarde/archivo.txt
